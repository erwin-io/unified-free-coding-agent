cmake_minimum_required(VERSION 3.20)
project(UnifiedAgentCpp VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(agent_math src/math.cpp)
target_include_directories(agent_math PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(agent_app src/main.cpp)
target_link_libraries(agent_app PRIVATE agent_math)
enable_testing()
add_executable(agent_tests tests/test_math.cpp)
target_link_libraries(agent_tests PRIVATE agent_math)
add_test(NAME agent_math_test COMMAND agent_tests)
